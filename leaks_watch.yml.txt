name: Leaks & Breaking News Watch

on:
  schedule:
    # يعمل كل ساعتين لاقتناص التسريبات والأخبار العاجلة قبل المواقع الشهيرة
    - cron: '0 */2 * * *'
  workflow_dispatch:

# إلغاء أي تشغيل متزامن لمنع نشر نفس التسريب مرتين عند تداخل الأوقات
concurrency:
  group: news-bot-leaks-single-run
  cancel-in-progress: true

jobs:
  run-leaks-watch:
    runs-on: ubuntu-latest

    permissions:
      contents: write

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: '3.10'

      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          pip install requests feedparser beautifulsoup4 pillow google-genai

      # نفس الوحدات المشتركة (config/utils/history_manager/fetcher/ai_handler/publisher)
      # + وحدات التسريبات الخاصة (leaks_config/leak_fetcher/leak_ai_handler/leak_publisher)
      - name: Execute Leaks Watch Script
        working-directory: news_bot
        env:
          GEMINI_API_KEY: ${{ secrets.GEMINI_API_KEY }}
          TELEGRAM_BOT_TOKEN: ${{ secrets.TELEGRAM_BOT_TOKEN }}
          TELEGRAM_CHAT_ID: ${{ secrets.TELEGRAM_CHAT_ID }}
          # اختياري: عيّن سرّاً منفصلاً هنا لاحقاً لو رغبت بقناة تسريبات مستقلة
          LEAK_TELEGRAM_CHAT_ID: ${{ secrets.LEAK_TELEGRAM_CHAT_ID }}
        run: python leaks_main.py

      - name: Commit and Push History
        run: |
          git config --global user.name "GitHub Action Bot"
          git config --global user.email "action@github.com"
          git add news_bot/posted_history.json || true
          git commit -m "Update posted leaks/news history [skip ci]" || true
          git push || true
